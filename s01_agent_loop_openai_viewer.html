#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
s01_agent_loop_openai.py - Agent Loop（OpenAI 版 / 只保留 run_bash 一个工具）

核心循环与 learn-claude-code s01 完全一致：

    while True:
        response = LLM(messages, tools)
        if 没有 tool_calls: break
        执行工具 -> 把 tool 结果追加进 messages -> 继续

    +----------+      +-------+      +---------+
    |   User   | ---> |  LLM  | ---> |  Tool   |
    |  prompt  |      |       |      | execute |
    +----------+      +---+---+      +----+----+
                          ^               |
                          |   tool_result |
                          +---------------+
                          (loop continues)

与 s01 原版的两点不同：
    1. 大模型改用 openai SDK 调用（chat.completions.create）。
    2. 工具描述用 tool_description_generator.generate_tool_description 自动生成，
       工具结果按 OpenAI 规范用 role="tool" 消息回传（参考《大模型的函数调用方法》）。

用法：
    pip install openai python-dotenv
    python s01_agent_loop_openai.py
"""

import json
import os
import subprocess
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

try:
    from tool_description_generator import generate_tool_description
except ImportError:
    sys.exit("[Error] 未找到 tool_description_generator.py，请把它放在本文件同级目录。")

from dotenv import load_dotenv
from openai import OpenAI

# Windows 控制台：切到 UTF-8 代码页 + 启用 ANSI 颜色，失败也无妨
if os.name == "nt":
    os.system("chcp 65001 > nul")
    os.system("")
# 统一用 UTF-8 输出并容错，避免 GBK 控制台下 print 抛 UnicodeEncodeError
for _stream in (sys.stdout, sys.stderr):
    try:
        _stream.reconfigure(encoding="utf-8", errors="replace")
    except Exception:
        pass

load_dotenv(override=True)

# ---------- LLM 客户端（OpenAI 兼容协议） ----------
# 默认对接 DeepSeek（课程环境）；换成官方 OpenAI 时设置 OPENAI_BASE_URL=https://api.openai.com/v1
client = OpenAI(
    api_key=os.getenv("OPENAI_API_KEY"),
    base_url=os.getenv("OPENAI_BASE_URL", "https://api.deepseek.com"),
)
MODEL = os.getenv("MODEL_ID", "deepseek-chat")

SYSTEM = (
    f"You are a coding agent at {os.getcwd()}. "
    "Use the bash tool to solve tasks. Act, don't explain. "
    "The shell is Windows cmd.exe: use dir / type / echo / findstr / copy / del, "
    "never Unix commands like ls / cat / grep / wc."
)

MAX_ROUNDS = 40  # 安全上限，防止模型不停调用工具烧钱


# 工具注册表：函数名 -> 真实函数（按 notebook 的做法，用 map 分发）
TOOL_MAP = {"run_bash": run_bash}

# 用 generate_tool_description 自动生成 OpenAI 规范的工具描述
TOOLS = [generate_tool_description(run_bash)]
def main():
    print("s01: Agent Loop (OpenAI)")
    print(f"model  = {MODEL}")
    print(f"base   = {client.base_url}")
    print(f"cwd    = {os.getcwd()}")
    print("输入问题并回车发送，输入 q 退出。\n")

    history = [{"role": "system", "content": SYSTEM}]
    while True:
        try:
            query = input("\033[36ms01 >> \033[0m")
        except (EOFError, KeyboardInterrupt):
            break
        if query.strip().lower() in ("q", "exit", ""):
            break

        history.append({"role": "user", "content": _sanitize(query)})
        answer = agent_loop(history)
        print(answer)
        print()


if __name__ == "__main__":
    if not os.getenv("OPENAI_API_KEY"):
        sys.exit("[Error] 缺少 OPENAI_API_KEY，请在 .env 中配置。")
    main()

def _decode(data: bytes) -> str:
    """按 utf-8 -> gbk 顺序解码命令输出，都失败则替换非法字符"""
    for enc in ("utf-8", "gbk"):
        try:
            return data.decode(enc)
        except UnicodeDecodeError:
            continue
    return data.decode("utf-8", errors="replace")
def _sanitize(text) -> str:
    """去掉 Windows 控制台可能引入的孤立代理字符（如 \\udcxx）

    这类字符会让 openai SDK 在 JSON 序列化时报
    UnicodeEncodeError: surrogates not allowed，必须在发往 API 前清洗。
    """
    text = "" if text is None else str(text)
    return text.encode("utf-8", errors="replace").decode("utf-8")
def run_bash(command: str) -> str:
    """执行一条 shell 命令并返回它的输出

    :param command: 要执行的 cmd 命令，例如 dir /b *.py、type a.txt、echo hi > a.txt
    :return: 命令的标准输出与错误输出（合并），无输出时返回占位文本
    """
    dangerous = ["rm -rf /", "sudo", "shutdown", "reboot", "> /dev/", "format "]
    if any(d in command for d in dangerous):
        return "Error: Dangerous command blocked"
    try:
        r = subprocess.run(
            command,
            shell=True,
            cwd=os.getcwd(),
            capture_output=True,
            timeout=120,
        )
        # 拿到原始字节后自己解码：Windows 中文输出多为 GBK，直接按 utf-8 解会得到替换符
        raw = (r.stdout or b"") + (r.stderr or b"")
        out = _decode(raw).strip()
        if not out:
            return f"(no output, exit code {r.returncode})"
        return out[:50000]
    except subprocess.TimeoutExpired:
        return "Error: Timeout (120s)"
    except (FileNotFoundError, OSError) as e:
        return f"Error: {e}"
def agent_loop(messages: list) -> str:
    """反复调用 LLM，直到它不再请求工具为止"""
    for _ in range(MAX_ROUNDS):
        response = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=TOOLS,
            tool_choice="auto",
            max_tokens=8000,
        )
        message = response.choices[0].message

        # 1) 追加 assistant 消息（有工具调用时必须带上 tool_calls）
        assistant_msg = {"role": "assistant", "content": _sanitize(message.content or "")}
        if message.tool_calls:
            assistant_msg["tool_calls"] = [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments,
                    },
                }
                for tc in message.tool_calls
            ]
        messages.append(assistant_msg)

        # 2) 没有工具调用 => 循环结束
        if not message.tool_calls:
            return message.content or ""

        # 3) 执行每个工具调用
        for tc in message.tool_calls:
            if message.content:
                print(message.content)

            func_name = tc.function.name
            try:
                func_args = json.loads(tc.function.arguments or "{}")
            except json.JSONDecodeError:
                func_args = {}

            if func_name not in TOOL_MAP:
                output = f"Error: Unknown tool {func_name}"
                print(f"\033[31m! {output}\033[0m")
            else:
                print(f"\033[33m$ {func_args.get('command', '')}\033[0m")
                try:
                    output = TOOL_MAP[func_name](**func_args)
                except Exception as e:  # 工具本身炸了也要把错误喂回模型
                    output = f"Error: {e}"
                print(output[:2000])

            # 4) 按 OpenAI 规范把结果回传
            messages.append(
                {
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": _sanitize(output),
                }
            )

    return "[Stopped] Reached max rounds limit."

代码解析

主程序入口

主程序入口
功能概述

该程序实现了一个基于 OpenAI 兼容协议（默认对接 DeepSeek）的 Agent Loop 智能体循环。核心思想是让大模型在“思考—调用工具—获取结果—继续思考”的闭环中自主完成任务。程序只注册了一个工具 run_bash，用于在 Windows cmd.exe 环境下执行 shell 命令，从而让模型能够读写文件、查看目录、运行脚本等。整个循环与 learn-claude-code s01 版本保持一致，区别在于改用 openai SDK 的 chat.completions.create 接口，并通过 tool_description_generator 自动生成符合 OpenAI 规范的工具描述，工具结果以 role=“tool” 的消息回传。

入口点与执行流程

程序入口位于文件末尾的 if __name__ == “__main__” 代码块。启动时首先检查环境变量 OPENAI_API_KEY 是否存在，缺失则通过 sys.exit 输出错误并终止。校验通过后调用 main() 函数。main() 打印模型名、base_url、当前工作目录等启动信息，随后进入一个外层交互循环：通过 input 读取用户输入（带青色 ANSI 提示符），遇到 EOFError 或 KeyboardInterrupt 时跳出，输入 q、exit 或空字符串时也退出。有效输入经 _sanitize 清洗后追加到 history 中作为 user 消息，然后调用 agent_loop(history) 执行内层智能体循环，最后打印返回的答案并空一行。

关键初始化步骤与配置

初始化阶段包含多项环境适配与配置。首先通过 sys.path.insert 将脚本所在目录加入模块搜索路径，确保能导入同级的 tool_description_generator 模块，导入失败则直接退出并提示。接着针对 Windows 平台执行 chcp 65001 切换 UTF-8 代码页并启用 ANSI 颜色支持，同时对 sys.stdout 和 sys.stderr 调用 reconfigure 设置 UTF-8 编码与 replace 错误处理，避免 GBK 控制台下打印中文或特殊字符时抛出 UnicodeEncodeError。随后调用 load_dotenv(override=True) 加载 .env 配置。LLM 客户端通过 OpenAI 类实例化，api_key 取自 OPENAI_API_KEY，base_url 默认指向 https://api.deepseek.com，可通过 OPENAI_BASE_URL 覆盖；模型 ID 由 MODEL_ID 环境变量决定，默认 deepseek-chat。SYSTEM 提示词动态嵌入当前工作目录，并明确告知模型 shell 为 Windows cmd.exe，要求使用 dir、type、echo、findstr 等命令而非 Unix 命令。MAX_ROUNDS 设为 40，作为防止模型无限调用工具导致费用失控的安全上限。

主要组件的创建与交互

程序包含几个核心组件。其一是 TOOL_MAP 字典，将工具名 run_bash 映射到真实函数，作为分发注册表。其二是 TOOLS 列表，通过 generate_tool_description(run_bash) 自动生成符合 OpenAI function calling 规范的工具 schema，供 LLM 调用时声明可用工具。其三是 client 客户端对象，负责与 LLM 服务通信。其四是 history 消息列表，以 system 消息初始化，贯穿整个会话，累积用户输入、模型回复、工具调用请求与工具执行结果，从而维持多轮上下文。这些组件在 agent_loop 中协同工作：agent_loop 将 history 与 TOOLS 一起发送给 LLM，解析返回的 tool_calls，通过 TOOL_MAP 找到对应函数执行，再把结果以 role=“tool” 消息追加回 history，形成闭环。

主循环逻辑

程序存在两层循环。外层是 main() 中的用户交互循环，负责读取输入、维护会话历史并调用内层循环，直到用户退出。内层是 agent_loop 中的智能体循环，其逻辑为：在不超过 MAX_ROUNDS 的轮次内，调用 client.chat.completions.create 传入 messages=history 与 tools=TOOLS；若响应中没有 tool_calls，则说明模型已给出最终答案，直接返回其 content 并结束；若有 tool_calls，则遍历每个调用，解析函数名与 JSON 参数，从 TOOL_MAP 取出对应函数执行，将执行结果序列化后以 {“role”: “tool”, “tool_call_id”: ..., “content”: ...} 形式追加到 history，同时把模型的 assistant 消息（含 tool_calls）也追加进历史，然后进入下一轮。该循环对应文档中 User→LLM→Tool→tool_result→LLM 的往复图示，直到模型不再请求工具或达到轮次上限为止。

健壮性与设计考量

代码在多个层面考虑了健壮性：导入失败、缺少 API Key 均给出明确错误并退出；Windows 控制台编码问题通过代码页切换与流重配置双重保障；输入处理捕获 EOFError 与 KeyboardInterrupt 以优雅退出；MAX_ROUNDS 限制防止失控调用；_sanitize 对用户输入做清洗。设计上采用工具注册表加自动描述生成的方式，使新增工具只需注册函数即可，具备一定扩展性。整体结构清晰，将配置、工具定义、循环逻辑与交互入口分离，便于教学与二次开发。

_decode
核心功能

该函数 _decode 的核心功能是将字节串（bytes）转换为字符串（str），并采用多编码回退策略来应对不同来源的命令输出。设计意图非常明确：在跨平台或跨语言环境下执行系统命令时，输出字节流的编码往往不确定（例如 Windows 中文环境常见 GBK，Linux 常见 UTF-8），因此需要一个健壮的解码函数，尽可能正确地还原文本内容，而不是直接抛出异常导致程序中断。

参数与返回值

函数接收一个名为 data 的 bytes 类型参数，返回一个 str 类型结果。类型注解清晰，便于静态检查和调用方理解接口契约。返回值始终为字符串，不会抛出解码异常，这体现了“容错优先”的设计原则。

解码策略与回退顺序

函数使用 for 循环遍历编码元组 ("utf-8", "gbk")，按顺序尝试解码。优先尝试 UTF-8 是因为它是现代系统和网络传输的事实标准，覆盖面最广；若失败则回退到 GBK，主要针对中文 Windows 环境下的命令行输出。这种顺序安排兼顾了通用性与本地化场景。

异常处理机制

在循环内部使用 try/except 捕获 UnicodeDecodeError。当某个编码无法正确解码字节序列时，捕获异常并通过 continue 进入下一次循环尝试下一个编码。这里只捕获特定的 UnicodeDecodeError，而不是宽泛的 Exception，属于精确异常处理，避免掩盖其他潜在错误（如类型错误），是良好的实践。

最终兜底方案

如果 UTF-8 和 GBK 都解码失败，函数执行最后的 return data.decode("utf-8", errors="replace")。这里使用 errors="replace" 参数，将无法解码的字节替换为 Unicode 替换字符（U+FFFD，即 �），保证函数一定能返回字符串而不抛出异常。这是一种“尽力而为”的降级策略，确保调用方始终能拿到可用的文本结果。

技术要点与设计权衡

整体实现体现了渐进式降级的思想：严格解码优先，宽松替换兜底。优点在于简洁、健壮、无外部依赖。潜在局限在于：GBK 解码失败后回退到 UTF-8 替换，可能产生乱码字符，但这是无法确定真实编码时的合理妥协；此外，某些字节序列在 UTF-8 和 GBK 下都能成功解码但含义不同（歧义编码），此时会优先采用 UTF-8 的结果，这是顺序策略带来的固有取舍。

_sanitize
核心功能

该函数 _sanitize 的核心职责是对输入文本进行 Unicode 清洗，移除或替换掉可能导致后续 JSON 序列化失败的非法字符，特别是 Windows 控制台环境下可能产生的孤立代理字符（lone surrogates，如 \udcxx 范围内的字符）。设计意图是作为一道防御性屏障，在文本发往 OpenAI API 之前保证其可被安全地 UTF-8 编码和 JSON 序列化。

问题背景

孤立代理字符是 UTF-16 编码体系中的残留物，它们本应与另一个代理字符配对组成完整的码点，但单独出现时无法映射到合法的 Unicode 码点。Python 的 str 类型允许存储这类字符，但当 openai SDK 尝试将其序列化为 JSON 并编码为 UTF-8 时，会抛出 UnicodeEncodeError: surrogates not allowed。Windows 控制台在读取或转换输入时容易产生这类字符，因此需要专门清洗。

实现细节

函数首先通过 text = "" if text is None else str(text) 做空值归一化处理：None 被转换为空字符串，其他类型统一转为字符串，保证后续操作的对象一定是 str 类型。这一步提升了函数的健壮性，使其可以接受任意输入而不抛出异常。

核心转换逻辑

关键语句 text.encode("utf-8", errors="replace").decode("utf-8") 采用“编码-解码”往返策略完成清洗。encode 时指定 errors="replace"，意味着遇到无法编码的字符（包括孤立代理字符）时，用替换字符（通常是 ? 或 U+FFFD）替代，而不是抛出异常；随后再 decode 回 str，得到的是不含非法代理字符的干净文本。这种往返方式简洁且能覆盖所有不可编码字符，不仅限于代理字符。

技术要点

使用 errors="replace" 而非 errors="ignore" 是一种权衡：replace 保留了字符位置的占位信息，避免文本因删除字符而发生语义粘连；ignore 则会直接丢弃非法字符。选择 replace 更符合“清洗”而非“删除”的语义，也便于调试时发现异常输入的存在。

返回值与类型

函数通过类型注解 -> str 明确返回字符串类型，且在所有分支下（包括 None 输入）都保证返回 str，调用方无需再做类型判断，接口契约清晰。

适用场景与局限

该函数适用于文本预处理管道中的最后一道清洗，尤其适合与外部 API 交互前的数据准备。需要注意的是，替换后的字符可能与原文语义有偏差，若业务对文本保真度要求极高，可能需要记录或告警被替换的内容；此外该方案针对的是编码层面的非法字符，并不处理其他类型的文本规范化问题（如 NFC/NFD 归一化）。

run_bash
核心功能

该函数封装了 subprocess.run 调用，用于执行一条 shell 命令并返回合并后的标准输出与错误输出。设计意图是为上层调用者（如 Agent 工具、自动化脚本）提供一个安全、可控、跨平台的命令执行入口，同时通过危险命令过滤、超时控制和输出截断来降低风险。

安全防护

函数首先定义了一个 dangerous 列表，包含 rm -rf /、sudo、shutdown、reboot、> /dev/、format 等字符串，并通过 any(d in command for d in dangerous) 做子串匹配。一旦命中即返回错误信息而不执行命令。这是一种黑名单式的简单防护，能拦截明显的破坏性命令，但存在明显局限：子串匹配容易被绕过（如 rm -fr /、大小写变体、变量拼接、base64 编码等），也无法覆盖 Windows 下的 del /f /s /q C:\ 等危险操作。因此它只能作为最低限度的护栏，不能替代真正的沙箱或权限隔离。

命令执行

核心调用为 subprocess.run(command, shell=True, cwd=os.getcwd(), capture_output=True, timeout=120)。shell=True 允许传入完整的命令行字符串（支持管道、重定向、通配符），这是该函数能执行 dir /b *.py、echo hi > a.txt 这类命令的前提，但也意味着命令会经由系统 shell 解释，进一步放大了注入风险。cwd 显式设置为当前工作目录，保证相对路径行为可预期。capture_output=True 等价于同时捕获 stdout 和 stderr。timeout=120 为命令设置了 120 秒上限，防止挂起进程长期占用资源。

输出处理

函数将 r.stdout 与 r.stderr 以字节形式拼接（用 or b"" 处理 None 情况），然后交给 _decode 辅助函数解码。注释明确指出 Windows 中文环境下输出多为 GBK，若直接按 UTF-8 解码会产生替换符，因此自行解码是必要的兼容处理。解码后调用 strip() 去除首尾空白；若结果为空，则返回包含退出码的占位文本 (no output, exit code N)，便于调用方判断命令是否成功。最后通过 out[:50000] 将输出截断到 5 万字符，避免超长输出撑爆上下文或内存。

异常处理

函数捕获两类异常：subprocess.TimeoutExpired 返回固定的超时错误文本；FileNotFoundError 与 OSError 返回带异常信息的错误字符串。这种设计保证函数在任何情况下都返回字符串而不抛出异常，符合工具函数“永不崩溃、始终可读”的契约。不过它没有捕获 subprocess.SubprocessError 的其他子类，也没有区分退出码非零但仍有输出的情况——后者会被正常返回，调用方需自行解析 exit code 占位文本或依赖输出内容判断。

设计权衡

整体来看，该函数在易用性与安全性之间做了折中：shell=True 带来便利但牺牲隔离性；黑名单过滤成本低但防护弱；输出截断与超时控制提升了健壮性；自解码解决了跨平台编码痛点。若要用于生产环境，建议改为 shell=False 加参数列表、使用白名单或命令解析器、在独立低权限用户/容器中执行，并将危险命令检测从子串匹配升级为结构化解析。

agent_loop
核心功能

该函数实现了一个典型的 Agent 循环（ReAct 风格），通过反复调用大语言模型（LLM）并执行其请求的工具，直到模型不再请求工具或达到最大轮数为止。其设计意图是构建一个可自主决策、调用外部工具并根据工具结果继续推理的智能体执行引擎。

整体流程

函数接收一个 messages 列表作为对话上下文，在 MAX_ROUNDS 次循环内不断向 LLM 发起请求。每轮将模型返回的 assistant 消息追加到上下文，若模型请求了工具则逐个执行并把结果以 tool 角色回传，然后进入下一轮；若模型不再请求工具，则直接返回其文本内容作为最终答案。

LLM 调用

通过 client.chat.completions.create 调用模型，传入 model、messages、tools 和 tool_choice="auto"，其中 tool_choice="auto" 表示由模型自行决定是否调用工具。max_tokens=8000 限制了单次响应的长度。response.choices[0].message 取出首个候选消息作为本轮模型输出。

assistant 消息构造

代码将模型返回的内容封装为 role 为 assistant 的字典，并对 content 调用 _sanitize 做清洗（可能用于去除非法字符或控制长度）。关键细节是：当存在 tool_calls 时，必须把工具调用信息按 OpenAI 规范序列化为包含 id、type 和 function（name、arguments）的结构一并附加，否则后续 tool 消息将无法与调用对应，导致 API 报错。

终止条件

在追加 assistant 消息后立即检查 message.tool_calls，若为空则说明模型已完成推理，直接返回 message.content（为空时返回空字符串）。这是循环的正常退出路径，保证最终答案能返回给调用方。

工具执行

对每个 tool_call，先通过 json.loads 解析 arguments，解析失败时降级为空字典 {}，避免因模型输出非法 JSON 而崩溃。随后检查工具名是否存在于 TOOL_MAP 中：不存在则生成 Unknown tool 错误信息；存在则调用对应函数并传入解析后的参数。工具执行被 try/except 包裹，即使工具内部抛出异常也会被捕获并转为 Error 字符串，从而把错误信息反馈给模型让其自我修正，而不是中断整个循环。

结果回传

每个工具执行结果都以 role 为 tool、携带对应 tool_call_id 的消息追加到 messages 中，content 同样经过 _sanitize 处理。tool_call_id 与之前的 tool_calls.id 一一对应，符合 OpenAI 的工具调用协议，确保模型能将结果与请求关联。

日志与可观测性

代码在工具执行前后打印了带 ANSI 颜色码的日志：红色（\033[31m）标记未知工具错误，黄色（\033[33m）打印即将执行的命令（取 func_args 中的 command 字段），并截断输出到前 2000 字符打印，便于调试和观察 Agent 行为。此外在存在 content 时也会打印模型文本。

轮数限制与兜底

循环受 MAX_ROUNDS 约束，防止模型陷入无限工具调用循环。若达到上限仍未收敛，函数返回 “[Stopped] Reached max rounds limit.” 作为兜底提示，保证函数总有返回值且不会死循环。

潜在风险与改进点

一是 messages 列表为原地修改，调用方会感知到上下文被追加，需注意副作用；二是 func_args.get('command', '') 假设参数中可能有 command 字段，属于特定工具约定，通用性有限；三是异常捕获过于宽泛（except Exception），可能掩盖严重错误；四是未对工具返回值的类型做统一处理，若返回非字符串可能影响 _sanitize 的行为。
