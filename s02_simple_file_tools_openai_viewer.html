#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
s02_simple_file_tools_openai.py - 文件工具（OpenAI 版 / 极简）

s01 的循环一行不改，只把「一个 bash」换成「四个文件工具 + 一张分发表」，
功能与 learn-claude-code s02 的 code.py 对齐（run_read / run_write / run_edit / run_glob）：

    +----------+      +-------+      +-------------------------+
    |   User   | ---> |  LLM  | ---> | Tool Dispatch           |
    |  prompt  |      |       |      | run_bash    -> 跑命令   |
    +----------+      +---+---+      | read_file   -> 读文件   |
                          ^          | write_file  -> 写文件   |
                          |          | edit_file   -> 补丁替换 |
                          +----------+ glob_files  -> 路径匹配 |
                         tool_result +-------------------------+

    + safe_path(): 文件工具一律限制在工作目录内
    + TOOL_MAP 分发，替换 s01 里硬编码的 run_bash 调用

用法：
    pip install openai python-dotenv
    python s02_simple_file_tools_openai.py
"""

import json
import os
import subprocess
import sys
from pathlib import Path

# tool_description_generator.py 可能在本目录，也可能在上一层目录
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

try:
    from tool_description_generator import generate_tool_description
except ImportError:
    sys.exit("[Error] 未找到 tool_description_generator.py，请把它放在本文件同级或上层目录。")

from dotenv import load_dotenv
from openai import OpenAI

# Windows 控制台：切到 UTF-8 代码页，避免 print 中文抛 UnicodeEncodeError
if os.name == "nt":
    os.system("chcp 65001 > nul")
    os.system("")
for _stream in (sys.stdout, sys.stderr):
    try:
        _stream.reconfigure(encoding="utf-8", errors="replace")
    except Exception:
        pass

load_dotenv(override=True)

# ---------- LLM 客户端（OpenAI 兼容协议） ----------
client = OpenAI(
    api_key=os.getenv("OPENAI_API_KEY"),
    base_url=os.getenv("OPENAI_BASE_URL", "https://api.deepseek.com"),
)
MODEL = os.getenv("MODEL_ID", "deepseek-chat")

WORKDIR = Path.cwd().resolve()

SYSTEM = (
    f"You are a coding agent at {WORKDIR}. Use tools to solve tasks. Act, don't explain. "
    "Prefer read_file/glob_files over bash, and edit_file over write_file. "
    "The shell is Windows cmd.exe: use dir / type / echo / findstr / copy / del, "
    "never Unix commands like ls / cat / grep / wc."
)

MAX_ROUNDS = 40  # 安全上限，防止模型不停调用工具烧钱


# 工具注册表：函数名 -> 真实函数
TOOL_MAP = {
    "run_bash": run_bash,
    "read_file": read_file,
    "write_file": write_file,
    "edit_file": edit_file,
    "glob_files": glob_files,
}

# 用 generate_tool_description 自动生成 OpenAI 规范的工具描述
TOOLS = [generate_tool_description(fn) for fn in TOOL_MAP.values()]
def main():
    print("s02: File Tools (OpenAI, simple) - " + ", ".join(TOOL_MAP))
    print(f"model  = {MODEL}")
    print(f"base   = {client.base_url}")
    print(f"cwd    = {WORKDIR}")
    print("输入问题并回车发送，输入 q 退出。\n")

    history = [{"role": "system", "content": SYSTEM}]
    while True:
        try:
            query = input("\033[36ms02 >> \033[0m")
        except (EOFError, KeyboardInterrupt):
            break
        if query.strip().lower() in ("q", "exit", ""):
            break

        history.append({"role": "user", "content": _sanitize(query)})
        answer = agent_loop(history)
        print(answer)
        print()


if __name__ == "__main__":
    if not os.getenv("OPENAI_API_KEY"):
        sys.exit("[Error] 缺少 OPENAI_API_KEY，请在 .env 中配置。")
    main()

def _decode(data: bytes) -> str:
    """按 utf-8 -> gbk 顺序解码命令输出，都失败则替换非法字符"""
    for enc in ("utf-8", "gbk"):
        try:
            return data.decode(enc)
        except UnicodeDecodeError:
            continue
    return data.decode("utf-8", errors="replace")
def _sanitize(text) -> str:
    """去掉 Windows 控制台可能引入的孤立代理字符（如 \\udcxx），否则 openai SDK 序列化会报错"""
    text = "" if text is None else str(text)
    return text.encode("utf-8", errors="replace").decode("utf-8")
def run_bash(command: str) -> str:
    """执行一条 shell 命令并返回它的输出

    :param command: 要执行的 cmd 命令，例如 dir /b *.py、type a.txt、echo hi > a.txt
    :return: 命令的标准输出与错误输出（合并），无输出时返回占位文本
    """
    dangerous = ["rm -rf /", "sudo", "shutdown", "reboot", "> /dev/", "format "]
    if any(d in command for d in dangerous):
        return "Error: Dangerous command blocked"
    try:
        r = subprocess.run(
            command,
            shell=True,
            cwd=str(WORKDIR),
            capture_output=True,
            timeout=120,
        )
        # 拿到原始字节后自己解码：Windows 中文输出多为 GBK，直接按 utf-8 解会得到替换符
        raw = (r.stdout or b"") + (r.stderr or b"")
        out = _decode(raw).strip()
        if not out:
            return f"(no output, exit code {r.returncode})"
        return out[:50000]
    except subprocess.TimeoutExpired:
        return "Error: Timeout (120s)"
    except (FileNotFoundError, OSError) as e:
        return f"Error: {e}"
def safe_path(p: str) -> Path:
    """解析路径并确保它还在工作目录内

    :param p: 相对或绝对路径
    :return: 落在工作目录内的绝对路径
    """
    path = (WORKDIR / p).resolve()
    if not str(path).lower().startswith(str(WORKDIR).lower()):
        raise ValueError(f"Path escapes workspace: {p}")
    return path
def read_file(path: str, offset: int = 1, limit: int = None) -> str:
    """读取文件内容，可用 offset/limit 分页，避免大文件撑爆上下文

    :param path: 文件路径，相对路径基于工作目录
    :param offset: 从第几行开始读，从 1 开始计数
    :param limit: 本次最多读多少行，不传则读到文件末尾
    :return: 文件内容；还有剩余行时给出下一次的 offset 提示
    """
    lines = safe_path(path).read_text(encoding="utf-8", errors="replace").splitlines()
    start = max(1, offset)
    end = len(lines) if not limit else min(len(lines), start + limit - 1)
    out = "\n".join(lines[start - 1:end])
    if end < len(lines):
        out += f"\n... ({len(lines) - end} more lines, continue with offset={end + 1})"
    return out
def write_file(path: str, content: str) -> str:
    """把完整内容写入文件（新建，或整体覆盖已有文件）

    :param path: 目标文件路径，父目录不存在时自动创建
    :param content: 文件的完整内容
    :return: 写入结果
    """
    file_path = safe_path(path)
    file_path.parent.mkdir(parents=True, exist_ok=True)
    file_path.write_text(content, encoding="utf-8")
    return f"Wrote {len(content)} bytes to {path}"
def edit_file(path: str, old_text: str, new_text: str) -> str:
    """按上下文做精确补丁替换：给出一段原文，替换成新片段（不用正则，只改这一处）

    :param path: 要修改的文件路径
    :param old_text: 待替换的原文片段，需与文件内容逐字符一致，可跨多行
    :param new_text: 替换后的新内容，传空串表示删除这段原文
    :return: 修改结果
    """
    file_path = safe_path(path)
    text = file_path.read_text(encoding="utf-8")
    if old_text not in text:
        return f"Error: text not found in {path}"
    file_path.write_text(text.replace(old_text, new_text, 1), encoding="utf-8")
    return f"Edited {path}"
def glob_files(pattern: str) -> str:
    """按 glob 模式匹配文件名或路径，改代码前先定位文件

    :param pattern: glob 模式，例如 **/*.py、*.md、src/**/*.ts；** 递归匹配任意层目录
    :return: 匹配到的相对路径列表，无匹配时返回 (no matches)
    """
    matches = sorted(str(p.relative_to(WORKDIR)) for p in WORKDIR.glob(pattern))
    return "\n".join(matches) if matches else "(no matches)"
def agent_loop(messages: list) -> str:
    """反复调用 LLM，直到它不再请求工具为止"""
    for _ in range(MAX_ROUNDS):
        response = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=TOOLS,
            tool_choice="auto",
            max_tokens=8000,
        )
        message = response.choices[0].message

        # 1) 追加 assistant 消息（有工具调用时必须带上 tool_calls）
        assistant_msg = {"role": "assistant", "content": _sanitize(message.content or "")}
        if message.tool_calls:
            assistant_msg["tool_calls"] = [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments,
                    },
                }
                for tc in message.tool_calls
            ]
        messages.append(assistant_msg)

        # 2) 没有工具调用 => 循环结束
        if not message.tool_calls:
            return message.content or ""

        # 3) 执行每个工具调用
        for tc in message.tool_calls:
            if message.content:
                print(message.content)

            func_name = tc.function.name
            try:
                func_args = json.loads(tc.function.arguments or "{}")
            except json.JSONDecodeError:
                func_args = {}

            if func_name not in TOOL_MAP:
                output = f"Error: Unknown tool {func_name}"
                print(f"\033[31m! {output}\033[0m")
            else:
                print(f"\033[33m$ {func_name} {str(func_args)[:300]}\033[0m")
                try:
                    output = TOOL_MAP[func_name](**func_args)
                except Exception as e:  # 工具本身炸了也要把错误喂回模型
                    output = f"Error: {e}"
                print(output[:2000])

            # 4) 按 OpenAI 规范把结果回传
            messages.append(
                {
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": _sanitize(output),
                }
            )

    return "[Stopped] Reached max rounds limit."

代码解析

主程序入口

主程序入口
功能概述

该程序是一个基于 OpenAI 兼容协议（默认对接 DeepSeek）的极简编码代理（coding agent）。它在 s01 版本“单一 bash 工具”的基础上，将工具集扩展为五个：run_bash、read_file、write_file、edit_file、glob_files，并通过一张工具分发表（TOOL_MAP）实现动态分发。程序以交互式 REPL 形式运行，用户输入自然语言任务，模型自主选择工具执行，直至给出最终答案。

入口点与启动流程

脚本以 if __name__ == "__main__": 作为入口。启动时首先检查环境变量 OPENAI_API_KEY 是否存在，缺失则通过 sys.exit 输出错误并终止。校验通过后调用 main() 函数。这种“先校验后执行”的结构避免了在缺少凭证时仍初始化客户端并进入循环。

模块导入与路径处理

程序在导入阶段做了两处关键处理。第一，将当前文件所在目录及其父目录插入 sys.path 头部，以便无论 tool_description_generator.py 位于同级还是上层目录都能被导入；若导入失败则立即 sys.exit 并提示用户放置位置。第二，针对 Windows 控制台，通过 chcp 65001 切换代码页，并对 sys.stdout 与 sys.stderr 调用 reconfigure 设置 UTF-8 编码与 replace 错误策略，防止中文输出触发 UnicodeEncodeError。这两步保证了跨平台与中文环境的健壮性。

配置加载与客户端初始化

使用 load_dotenv(override=True) 从 .env 加载环境变量，override=True 表示环境文件中的值优先于已存在的系统环境变量。随后创建 OpenAI 客户端，api_key 取自 OPENAI_API_KEY，base_url 默认回退到 https://api.deepseek.com，模型 ID 默认 deepseek-chat。WORKDIR 通过 Path.cwd().resolve() 获取当前工作目录的绝对路径，作为后续文件工具的沙箱根目录。

系统提示词设计

SYSTEM 常量拼接了工作目录路径，并给出行为约束：以行动代替解释、优先使用 read_file/glob_files 而非 bash、优先 edit_file 而非 write_file。同时明确告知模型 shell 为 Windows cmd.exe，应使用 dir/type/echo/findstr 等命令而非 Unix 命令。这一提示词直接影响了模型选择工具与生成命令的方式，是保证工具调用正确性的关键配置。

工具注册与描述生成

TOOL_MAP 是一个字典，将工具名称字符串映射到对应的真实函数对象，作为运行时分发的依据。TOOLS 列表则通过对 TOOL_MAP.values() 中的每个函数调用 generate_tool_description 自动生成符合 OpenAI function calling 规范的描述结构。这种“注册表 + 自动描述”的设计使得新增工具只需在 TOOL_MAP 中添加一项，描述与分发自动同步，降低了维护成本。

主循环逻辑

main() 首先打印启动横幅，包括工具列表、模型、base_url、工作目录及使用说明。随后初始化 history 列表，首条为 system 消息。进入 while True 循环：通过 input 读取用户输入（带青色 ANSI 提示符），捕获 EOFError 与 KeyboardInterrupt 以优雅退出；若输入为 q、exit 或空字符串则跳出循环。否则将经过 _sanitize 处理的用户消息追加到 history，调用 agent_loop(history) 获取回答并打印，随后输出空行分隔。循环持续直到用户主动退出。

代理循环与安全上限

agent_loop 函数（代码中未展示实现，但由调用方式可推断）负责与模型多轮交互：发送 history 给模型，解析返回的 tool_calls，通过 TOOL_MAP 分发执行对应工具，将 tool_result 追加回 history 并再次请求模型，直至模型返回纯文本答案。MAX_ROUNDS = 40 作为安全上限，防止模型陷入无限工具调用循环导致费用失控，是成本控制的重要保护机制。

安全边界设计

文档字符串提到 safe_path() 将所有文件工具限制在工作目录内，这是防止路径穿越攻击（如 ../../etc/passwd）的核心防护。结合 WORKDIR 的绝对路径解析，文件读写与 glob 匹配均被约束在项目沙箱中，避免代理误操作或恶意操作影响系统其他区域。

整体架构评价

该程序采用“注册表分发 + 自动工具描述 + 交互式 REPL + 多轮代理循环”的经典 agent 架构，结构清晰、扩展性好。通过环境变量与 .env 实现配置外置，通过 sys.path 注入与编码重配置提升可移植性，通过 MAX_ROUNDS 与 safe_path 提供成本与安全双重保障。整体代码简洁，符合“极简文件工具代理”的设计目标。

_decode
函数功能

该函数 _decode 的核心功能是将字节串（bytes）转换为字符串（str），并采用多编码回退策略来提升解码成功率。设计意图是应对命令输出等外部数据来源编码不确定的场景：在中文环境下，命令输出可能是 UTF-8 编码，也可能是 GBK 编码，函数通过依次尝试两种编码来尽可能正确地还原文本内容。

参数与返回值

函数接收一个 bytes 类型的参数 data，返回一个 str 类型的结果。类型注解明确了输入输出契约，便于静态检查和调用方理解。返回值始终为字符串，不会抛出解码异常，属于“永不失败”的容错型接口。

编码回退策略

函数使用 for 循环遍历编码元组 ("utf-8", "gbk")，按顺序尝试解码。UTF-8 优先是因为它是现代系统与跨平台场景的主流编码，覆盖面广；GBK 作为后备，用于兼容中文 Windows 环境下的传统输出。这种顺序设计兼顾了通用性与本地化兼容性。

异常处理机制

在 try 块中调用 data.decode(enc)，若抛出 UnicodeDecodeError 则通过 continue 跳过当前编码，进入下一次循环尝试。这里只捕获 UnicodeDecodeError 这一特定异常，而非宽泛的 Exception，体现了精确异常处理的原则，避免掩盖其他潜在错误（如类型错误）。

最终兜底方案

当两种编码均解码失败时，循环结束，函数执行 data.decode("utf-8", errors="replace")。该兜底使用 errors="replace" 参数，将无法解码的字节替换为替换字符（通常是 U+FFFD），保证函数在任何输入下都能返回字符串而不抛出异常，提升了程序的健壮性。

设计权衡

该实现以“尽量正确、绝不崩溃”为优先目标。其局限在于：若数据实际为 GBK 编码但恰好能被 UTF-8 部分解码（较少见但存在），可能产生乱码而非回退到 GBK；此外 errors="replace" 会丢失原始字节信息。对于日志、终端输出等容错场景，这些权衡是合理且常见的。

代码风格

函数简洁紧凑，使用元组遍历替代重复的 try-except 块，减少了代码冗余。文档字符串清晰说明了回退顺序与兜底行为，可读性良好，符合 Python 惯用写法。

_sanitize
核心功能

该函数用于清洗字符串中的非法 Unicode 字符，特别是 Windows 控制台环境下可能产生的孤立代理字符（lone surrogate，如 \udcxx）。这类字符在 Python 内部可以存在于 str 对象中，但无法被 UTF-8 正常编码，会导致 openai SDK 在 JSON 序列化时抛出 UnicodeEncodeError 或类似异常。函数通过一次编码-解码往返，将不可编码的字符替换为安全字符，从而保证后续序列化流程稳定。

设计意图

函数名以下划线开头，表明它是模块内部使用的私有工具函数，不对外暴露。其设计目标单一且明确：作为数据发送给外部 SDK 之前的防御性清洗层。注释直接点明了问题来源（Windows 控制台）和后果（openai SDK 序列化报错），说明这是一个针对特定运行时环境的兼容性补丁，而非通用的文本处理工具。

空值与类型归一化

第一行 text = "" if text is None else str(text) 承担两个职责：一是将 None 转换为空字符串，避免后续 encode 调用抛出 AttributeError；二是对非字符串输入（如数字、对象）调用 str() 进行强制转换，保证函数对任意输入类型都具有鲁棒性。这种写法使函数可以安全地用在可能传入 None 或混合类型的字段上，例如从日志、终端输出或外部接口获取的文本。

编码解码往返机制

核心实现 text.encode("utf-8", errors="replace").decode("utf-8") 利用了 UTF-8 编解码器对代理字符的处理规则。孤立代理字符（U+D800 至 U+DFFF 范围内未配对的值）在 UTF-8 中是非法的，encode 时配合 errors="replace" 会将其替换为问号字符（?，即 U+003F），而不是抛出异常。随后 decode("utf-8") 将字节序列还原为合法的 str。由于替换发生在编码阶段，解码阶段不会遇到非法字节，因此无需再指定 errors 参数。

技术要点与边界

该方案的关键在于 errors="replace" 参数的选择：若使用默认的 errors="strict"，encode 会直接抛出 UnicodeEncodeError；若使用 errors="ignore"，则会静默丢弃字符，可能造成信息丢失且不易察觉。replace 策略在保证不崩溃的同时保留了字符位置的可见痕迹。需要注意的是，该函数只处理孤立代理字符这一类问题，对于其他控制字符（如 \x00、ANSI 转义序列）并不会清除，因此它并非通用的输入净化函数。此外，编码解码往返对正常文本是无损的，性能开销为一次内存拷贝，对于常规长度的文本可以忽略。

使用场景

典型调用位置是在构造 API 请求体之前，对用户输入、终端捕获内容或日志文本统一过一遍 _sanitize，确保送入 openai SDK 的 JSON 序列化环节不会因编码问题中断。这种"边界清洗"模式在跨平台、跨进程的数据传递中是一种常见且低成本的防御性编程实践。

run_bash
核心功能

该函数 run_bash 的核心功能是封装 subprocess.run，对外提供一个受控的 shell 命令执行入口：接收一条命令字符串，在指定的工作目录下执行，并将标准输出与标准错误合并后以字符串形式返回。其设计意图是让上层调用者（如 AI Agent、自动化脚本或工具层）能够方便地执行系统命令并获取结果，同时通过危险命令过滤、超时控制和输出截断等手段降低执行风险。

安全防护机制

函数开头定义了一个 dangerous 列表，包含 rm -rf /、sudo、shutdown、reboot、> /dev/、format 等高风险片段，并通过 any(d in command for d in dangerous) 做子串匹配。一旦命中即直接返回错误提示，不进入执行流程。需要注意的是，这种基于子串的黑名单过滤属于较弱的防护手段，容易被绕过（例如使用变量拼接、路径变形、大小写变体或等价命令），因此它更适合作为“防误操作”的第一道门槛，而非严格的安全沙箱。

命令执行细节

实际执行使用 subprocess.run，并设置 shell=True，意味着命令会交由系统 shell 解释执行，从而支持管道、重定向、通配符等 shell 特性，这也是文档字符串中示例（如 echo hi > a.txt）能够工作的前提。cwd=str(WORKDIR) 将工作目录固定到预定义的 WORKDIR，保证命令在受控目录下运行，避免污染其他路径。capture_output=True 捕获 stdout 与 stderr，timeout=120 设置 120 秒超时，防止命令长时间挂起阻塞调用方。

输出解码处理

代码没有直接使用 r.stdout.decode()，而是将 stdout 与 stderr 的原始字节相加后交给自定义的 _decode 函数处理。注释明确指出原因：Windows 中文环境下命令输出多为 GBK 编码，若直接按 UTF-8 解码会产生替换符（乱码）。这说明 _decode 很可能实现了多编码尝试（如先 UTF-8 再 GBK/GB18030）的容错解码逻辑，体现了对跨平台、跨编码场景的细致考虑。

输出规范化与截断

解码后调用 .strip() 去除首尾空白；若结果为空，则返回包含退出码的占位文本 (no output, exit code N)，便于调用方区分“命令成功但无输出”与“执行失败”的情况。非空输出则通过 out[:50000] 截断到 5 万字符，防止超长输出（如大文件内容、日志）撑爆上下文或内存，这对面向大模型的工具调用场景尤为重要。

异常处理

函数通过 try/except 覆盖了两类异常：subprocess.TimeoutExpired 返回固定的超时错误文本；FileNotFoundError 与 OSError 则返回包含异常信息的错误字符串。所有异常都被转换为字符串返回值而非向上抛出，使得该函数具有“永不抛异常”的契约，调用方无需额外 try 包裹，接口更稳健。不过这也意味着调用方只能通过返回值前缀 Error: 来判断失败，缺乏结构化的错误类型区分。

整体评价

整体实现简洁实用，兼顾了安全性（黑名单+超时）、可用性（shell 特性、工作目录固定）、健壮性（编码容错、异常兜底）与资源控制（输出截断）。主要改进空间在于：危险命令检测可改为更严格的白名单或参数化执行；错误返回可考虑结构化（如返回字典或自定义异常）；以及 shell=True 本身带来的注入风险在面向不可信输入时应谨慎评估。

safe_path
核心功能

该函数 safe_path 的核心功能是将用户传入的路径字符串解析为绝对路径，并强制校验解析后的路径必须位于预定义的工作目录 WORKDIR 之内，否则抛出异常。其设计意图是防止路径穿越攻击（Path Traversal），确保程序只能访问受限的工作区，属于典型的安全边界防护函数。

路径解析机制

函数首先通过 WORKDIR / p 使用 pathlib 的路径拼接运算符将工作目录与输入路径组合。当 p 为相对路径时，会相对于 WORKDIR 解析；当 p 为绝对路径时，pathlib 的 / 运算符会直接采用右侧的绝对路径，从而覆盖左侧的 WORKDIR。随后调用 resolve() 方法，该方法会规范化路径、消除 . 和 .. 等相对片段，并解析符号链接，得到最终的绝对路径。这一步是安全校验的关键前提，因为只有规范化后的路径才能真实反映最终访问位置。

安全校验逻辑

校验通过比较字符串前缀实现：将解析后的路径与 WORKDIR 都转为字符串并统一转为小写后，检查前者是否以后者开头。使用 lower() 是为了在大小写不敏感的文件系统（如 Windows、macOS 默认）上避免因大小写差异导致的误判。若不满足前缀条件，则抛出 ValueError，提示路径逃逸出工作区。

潜在缺陷与风险

该实现存在若干安全隐患。第一，前缀匹配方式不严谨：若 WORKDIR 为 /home/user/work，而解析路径为 /home/user/workspace_evil，字符串同样以 /home/user/work 开头，会被错误放行，应改用 Path.is_relative_to 或比较路径组件。第二，大小写处理在大小写敏感的文件系统（如 Linux ext4）上可能引入误判，将本应不同的路径视为相同。第三，resolve() 默认在 Python 3.6+ 中对不存在的路径也会尽力解析，但符号链接解析行为可能因平台和版本而异，存在 TOCTOU（检查与使用时间差）风险，即校验后路径可能被替换。第四，未处理 WORKDIR 本身未规范化的情况，若 WORKDIR 含 .. 或符号链接，前缀比较可能失效。

参数与返回值

参数 p 为字符串类型的相对或绝对路径。返回值为 pathlib.Path 对象，表示落在工作目录内的绝对路径。函数通过抛出 ValueError 表达校验失败，调用方需捕获处理。

改进建议

建议使用 path.is_relative_to(WORKDIR.resolve()) 进行结构化比较以替代字符串前缀匹配；对 WORKDIR 预先做一次 resolve() 并缓存；在安全敏感场景下考虑使用 os.open 配合 O_NOFOLLOW 或文件描述符方式规避 TOCTOU；同时明确大小写策略以匹配目标文件系统语义。

read_file
核心功能

该函数实现了一个带分页能力的文本文件读取工具，主要设计意图是避免一次性将超大文件全部载入上下文（例如在 LLM 工具调用或交互式终端场景中），通过 offset/limit 参数按行分片读取，并在还有剩余内容时提示下一次读取的起始位置。

参数设计

path 为文件路径，注释说明相对路径基于工作目录；offset 表示起始行号且从 1 开始计数，默认值为 1；limit 表示本次最多读取的行数，默认 None 表示读到文件末尾。返回值是字符串形式的文件内容，可能附带续读提示。函数签名使用类型注解，可读性较好。

路径安全处理

代码通过 safe_path(path) 对传入路径做了一层封装处理（该函数未在此处给出），推测用于路径规范化、限制在工作目录内或防止目录穿越等安全校验。这是将安全策略与业务逻辑解耦的常见做法，但具体安全性依赖 safe_path 的实现，需结合其定义评估。

文件读取细节

使用 read_text 一次性读取整个文件，指定 encoding="utf-8" 并以 errors="replace" 容错，遇到无法解码的字节会替换而非抛异常，提升了鲁棒性。随后用 splitlines() 按行切分，能正确处理不同换行符（\n、\r\n 等）且不保留行尾换行符。需要注意：虽然对外表现为“分页”，但底层仍是一次性读入整个文件再切片，因此对内存占用并无优化，仅优化了返回给调用方的内容体积。

分页边界计算

start = max(1, offset) 保证起始行不小于 1，防止负数或 0 导致异常切片；end 在未指定 limit 时取总行数，否则取 min(len(lines), start + limit - 1)，既实现了“读 limit 行”的语义，又防止越界。这里 limit 为 0 或负数时会被当作假值走“读到末尾”分支，属于边界行为，调用方需注意。

切片与拼接

通过 lines[start - 1:end] 取出目标行区间（列表下标从 0 开始，故减 1），再用 "\n".join 重新拼接为字符串。这种基于列表切片的实现简洁高效，且天然兼容 offset 超出文件长度的情况（此时返回空字符串）。

续读提示机制

当 end < len(lines) 时，在结果末尾追加形如 "... (N more lines, continue with offset=M)" 的提示，其中 N 为剩余行数、M 为下一次应使用的 offset（end + 1）。该设计让调用方（尤其是自动化代理）能自主决定是否继续分页读取，形成可迭代的读取协议。

潜在改进点

一是大文件场景下可改用逐行迭代（如 itertools.islice）实现真正的流式分页以降低内存；二是 limit 为 0 或负数时语义不明确，建议显式校验；三是续读提示混入返回内容，若调用方需要纯文本内容可能造成干扰，可考虑通过返回值元组或独立字段区分内容与元信息。

write_file
核心功能

该函数实现了一个带安全校验的文件写入操作，其核心职责是将给定的字符串内容完整写入指定路径的文件中。函数采用“整体覆盖”语义，即无论目标文件是否已存在，都会用新内容替换原有内容，而非追加写入。这种设计意图通常用于配置文件生成、代码文件输出、模板渲染结果落盘等场景，强调写入结果的确定性和幂等性。

参数设计

函数接收两个参数：path 为目标文件路径字符串，content 为待写入的完整内容。返回值是一个描述写入结果的字符串，包含写入的字节数（实际为字符数，见下文分析）和原始路径。这种返回设计便于调用方记录日志或向用户反馈操作结果，同时保持了函数无副作用输出的简洁性。

路径安全处理

函数首先调用 safe_path(path) 对传入路径进行安全化处理，这是关键的防御性设计。虽然 safe_path 的具体实现未在此代码中展示，但从命名和调用位置可以推断，它可能负责路径规范化、防止路径穿越攻击（如 ../ 逃逸）、限制写入范围等安全职责。将安全校验前置到所有文件操作之前，体现了“先验证后执行”的安全编程原则。

目录自动创建

通过 file_path.parent.mkdir(parents=True, exist_ok=True) 实现父目录的自动创建。parents=True 表示递归创建所有不存在的上级目录，exist_ok=True 表示目录已存在时不抛出异常。这一设计使函数对调用方更加友好，调用者无需预先关心目录结构是否存在，降低了使用门槛，也避免了因目录缺失导致的写入失败。

编码与写入

使用 file_path.write_text(content, encoding="utf-8") 完成实际写入。显式指定 UTF-8 编码是重要的技术细节，它避免了依赖系统默认编码（在不同平台上可能为 GBK、ASCII 等）带来的乱码风险，保证了跨平台行为的一致性。write_text 是 pathlib.Path 提供的便捷方法，内部会以文本模式打开文件并写入，同时自动处理文件的关闭。

返回值语义

返回字符串 f"Wrote {len(content)} bytes to {path}" 存在一个值得注意的细节：len(content) 计算的是字符串的字符数，而非字节数。对于包含多字节字符（如中文）的内容，字符数与 UTF-8 编码后的实际字节数并不相等。因此返回值中的 "bytes" 表述在严格意义上不够准确，可能对调用方产生轻微误导。若需精确字节数，应使用 len(content.encode("utf-8"))。

异常处理

函数未显式捕获任何异常，写入失败（如权限不足、磁盘空间不足、路径非法等）时会将异常向上抛出。这种设计将错误处理的责任交给调用方，保持了函数的单一职责，符合 Python 的 EAFP 风格，但也要求调用方具备相应的异常处理意识。

整体评价

该函数结构清晰、职责单一，通过安全路径校验、目录自动创建、显式编码指定等手段，实现了一个健壮且易用的文件写入工具。主要改进空间在于返回值中字节数的准确计算，以及可考虑在文档中明确说明覆盖语义和可能的异常类型，以进一步提升接口的严谨性。

edit_file
核心功能

该函数实现了一个基于上下文精确匹配的文件内容替换工具，属于典型的“补丁式”编辑操作。其设计意图是让调用者通过提供一段与文件内容逐字符一致的原文片段（old_text），将其替换为新的片段（new_text），从而实现对文件的定点修改，避免使用正则表达式带来的复杂性和误匹配风险。

参数设计

函数接收三个参数：path 表示目标文件路径；old_text 是待替换的原文片段，要求与文件内容逐字符一致且可跨多行；new_text 是替换后的新内容，传入空字符串时表示删除该段原文。返回值统一为字符串形式的结果信息，成功时返回 “Edited {path}”，失败时返回以 “Error:” 开头的错误描述，这种设计便于调用方通过字符串前缀判断执行状态。

路径安全处理

函数首先调用 safe_path(path) 对传入路径进行安全校验与规范化，这通常用于防止路径穿越攻击（如包含 “../” 的恶意路径）或确保路径位于允许的工作目录内。该步骤将原始字符串路径转换为受控的 Path 对象 file_path，是安全边界的第一道防线。

文件读取与编码

使用 file_path.read_text(encoding=“utf-8”) 以 UTF-8 编码一次性读取整个文件内容到内存中的 text 变量。显式指定编码避免了平台默认编码差异导致的问题，保证跨平台行为一致；同时整文件读入的方式适合中小型文本文件的编辑场景。

存在性校验

通过 if old_text not in text 判断原文片段是否存在于文件内容中，若不存在则直接返回错误信息 “Error: text not found in {path}”，不进行任何写入操作。这一前置校验保证了操作的原子性语义——要么完整替换，要么完全不改动文件，避免产生部分修改的中间状态。

精确单次替换

核心替换逻辑使用 text.replace(old_text, new_text, 1)，第三个参数 1 是关键：它限制只替换第一次出现的匹配项。这与函数文档中“只改这一处”的承诺一致，防止当 old_text 在文件中多次出现时误改其他位置，体现了“按上下文精确定位”的设计思想。

写回与编码一致性

替换完成后通过 file_path.write_text(..., encoding=“utf-8”) 将新内容写回原文件，编码与读取时保持一致，确保不会因编码转换导致内容损坏。注意该写入方式是整体覆盖而非增量修改，因此不具备事务性或备份机制，若写入过程中断可能造成文件损坏。

技术要点与局限

该函数采用“读-改-写”模式，实现简洁但存在若干值得注意的特性：其一，非原子操作，读取与写入之间存在时间窗口，并发场景下可能丢失其他进程的修改；其二，old_text 必须逐字符精确匹配（包括空白与换行），对格式敏感；其三，未处理文件不存在、权限不足等异常，这些异常会以 Python 异常形式向上抛出而非转换为错误字符串返回，调用方需自行捕获处理。

glob_files
核心功能

该函数实现了一个基于 glob 模式的文件路径匹配工具，用于在工作目录（WORKDIR）下按指定模式查找文件或目录，并返回匹配结果的相对路径列表。其设计意图是作为代码修改前的辅助定位工具，帮助开发者快速找到目标文件，避免盲目修改代码。

参数设计

函数接收单一参数 pattern，类型为字符串，表示 glob 匹配模式。文档字符串中给出了多种示例，如 **/*.py 表示递归匹配所有 Python 文件，*.md 表示当前目录下的 Markdown 文件，src/**/*.ts 表示 src 目录下任意层级的 TypeScript 文件。其中 ** 是 glob 语法中用于递归匹配任意层目录的关键符号，体现了对深层目录结构的支持能力。

核心实现

函数体由一行生成式表达式构成：通过 WORKDIR.glob(pattern) 获取匹配的 Path 对象迭代器，对每个路径对象调用 relative_to(WORKDIR) 转换为相对于工作目录的路径，再用 str() 转为字符串，最后通过 sorted() 进行字典序排序。排序保证了输出结果的稳定性和可预测性，便于阅读和比对。

返回值处理

函数使用条件表达式决定返回值：当 matches 列表非空时，用换行符 "\n" 将各路径连接成单个字符串返回；当无匹配结果时，返回固定字符串 "(no matches)"。这种设计将列表统一转换为字符串输出，便于直接打印或嵌入到文本响应中，同时用明确的占位提示区分"无匹配"与"空结果"两种情况。

依赖与上下文

函数依赖外部定义的全局变量 WORKDIR，该变量应为 pathlib.Path 类型对象，代表工作目录的根路径。这种设计将目录范围限定在 WORKDIR 之内，既保证了相对路径输出的简洁性，也隐含了安全边界——匹配范围不会超出工作目录。但这也意味着函数不具备独立性，其行为强依赖于 WORKDIR 的正确初始化。

技术要点

一是使用 pathlib 的 glob 方法而非 glob 模块，前者返回 Path 对象，便于链式调用 relative_to 等方法；二是 relative_to 的使用确保了输出路径的可读性和可移植性，避免暴露绝对路径；三是 sorted 排序使结果具有确定性，有利于测试和结果复现；四是文档字符串明确提示"改代码前先定位文件"，体现了该函数在开发工作流中的定位价值。

潜在局限

函数未对 pattern 参数做合法性校验，非法模式可能引发异常；未处理 WORKDIR 不存在或匹配到目录的情况；返回值为单一字符串而非列表，调用方若需结构化数据需自行拆分。此外，若匹配结果极多，拼接成单个字符串可能较长，但考虑到其定位辅助用途，这一取舍是合理的。

agent_loop
核心功能

该函数实现了一个典型的 Agent 循环（ReAct 风格），通过反复调用大语言模型（LLM）并执行其请求的工具，直到模型不再请求工具或达到最大轮数为止。它是工具调用型 Agent 的主控流程，负责维护对话消息列表、解析工具调用、执行工具并将结果回传给模型。

设计意图

函数采用“思考-行动-观察”的迭代模式：模型先输出内容或工具调用，代码执行工具并把结果作为 tool 消息追加，再让模型基于新信息继续推理。通过 MAX_ROUNDS 限制循环次数，防止模型陷入无限工具调用循环，保证系统可控性。

循环控制

使用 for _ in range(MAX_ROUNDS) 作为外层循环，每轮调用一次 LLM。循环的退出条件有两个：一是模型返回的消息不含 tool_calls，此时直接返回模型内容；二是达到最大轮数，返回 “[Stopped] Reached max rounds limit.” 提示。这种双重退出机制兼顾了正常结束与异常保护。

LLM 调用

通过 client.chat.completions.create 发起请求，传入 model、messages、tools、tool_choice=“auto” 和 max_tokens=8000。tool_choice=“auto” 表示由模型自行决定是否调用工具，max_tokens 限制单次响应长度，避免过长输出。

assistant 消息构造

代码将模型响应转换为字典形式追加到 messages。关键细节是：当存在 tool_calls 时必须把工具调用信息一并写入 assistant 消息，且格式需符合 OpenAI 规范（包含 id、type、function.name、function.arguments）。这是后续 tool 消息能正确关联的前提。content 通过 _sanitize 处理，防止空值或非法字符。

终止判断

在追加 assistant 消息后立即检查 message.tool_calls，若为空则说明模型已完成推理，直接返回 message.content。这一判断放在工具执行之前，确保无工具调用时不会进入执行分支。

工具执行

遍历每个 tool_call，先解析 arguments（JSON 字符串），解析失败时降级为空字典 {}，避免因参数格式错误导致崩溃。随后检查工具名是否存在于 TOOL_MAP 中，未知工具返回错误字符串而非抛出异常，保证循环继续。

异常处理

工具执行被 try/except 包裹，任何异常都会被捕获并转换为 “Error: {e}” 字符串。设计意图是“工具本身炸了也要把错误喂回模型”，让模型有机会根据错误信息调整策略，而不是让整个 Agent 崩溃。这是健壮性的重要体现。

结果回传

每个工具执行结果都以 {“role”: “tool”, “tool_call_id”: tc.id, “content”: _sanitize(output)} 的形式追加到 messages。tool_call_id 必须与 assistant 消息中的 id 对应，这是 OpenAI 工具调用协议的要求，确保模型能将结果与请求匹配。

日志与调试

代码使用 ANSI 颜色码打印调试信息：红色（\033[31m）标记未知工具错误，黄色（\033[33m）标记工具调用及参数（截断至 300 字符），工具输出截断至 2000 字符。这些打印有助于开发者观察 Agent 行为，但属于副作用输出，生产环境可能需要改为日志系统。

潜在问题

一是 message.content 在工具循环内被重复打印（每个 tool_call 都判断一次），若模型同时返回内容和多个工具调用会造成重复输出；二是 _sanitize 的具体实现未展示，其行为影响消息合法性；三是未处理 API 调用本身的异常（如网络错误、限流），一旦 client.chat.completions.create 抛错，整个循环会中断；四是 MAX_ROUNDS 和 TOOL_MAP 等为外部全局变量，耦合度较高，不利于测试与复用。
