#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
s020_agent_loop_openai_toolclass.py - Agent Loop（OpenAI 版 / 一个工具 = 一个类）

核心循环与 s01 完全一致：

    while True:
        response = LLM(messages, tools)
        if 没有 tool_calls: break
        执行工具 -> 把 tool 结果追加进 messages -> 继续

    +----------+      +-------+      +---------+
    |   User   | ---> |  LLM  | ---> |  Tool   |
    |  prompt  |      |       |      | execute |
    +----------+      +---+---+      +----+----+
                          ^               |
                          |   tool_result |
                          +---------------+
                          (loop continues)

与 s01 的唯一区别：工具不再是模块级函数，而是一个类对应一个工具
（写法参考 s020/simple_react_agent.py 的 BaseTool + ToolRegistry）
    1. BaseTool：父类，子类只写 name + execute()；to_openai_schema() 内部调用
       tool_description_generator.generate_tool_description，从 execute 的类型注解
       和 docstring（:param xxx:）自动生成 description / parameters，与 s01 一致。
    2. RunBashTool(BaseTool)：唯一的工具，就是 s01 里的 run_bash，
       共享配置（workdir / timeout / 危险命令黑名单）放在 self 上。
    3. ToolRegistry：register(工具实例) / get / execute / get_definitions，
       出错时返回 "Error: ..." 文本，照样喂回模型。
    4. agent_loop 与 s01 同构，只是分发从 TOOL_MAP 换成 REGISTRY。

扩展方式：新写一个 class XxxTool(BaseTool) -> REGISTRY.register(XxxTool())，
循环和下面两行常量都不用动。

用法：
    pip install openai python-dotenv
    python s020_agent_loop_openai_toolclass.py
"""

import json
import os
import subprocess
import sys
from datetime import datetime
from typing import Any, Dict, List, Optional

# 系统提示词里给模型看的「星期几」，中文比 %A 的英文更省 token 也更直观
_WEEKDAYS = ("星期一", "星期二", "星期三", "星期四", "星期五", "星期六", "星期日")

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

    print(f"model  = {MODEL}")
    print(f"base   = {client.base_url}")
    print(f"cwd    = {os.getcwd()}")
    print(f"tools  = {', '.join(TOOL_MAP) or '(none)'}")
    print("输入问题并回车发送，输入 q 退出。\n")

    while True:
        try:
            query = input("\033[36ms020 >> \033[0m")
        except (EOFError, KeyboardInterrupt):
            break
        if query.strip().lower() in ("q", "exit", ""):
            break

        history.append({"role": "user", "content": _sanitize(query)})
        answer = agent_loop(history)
        print(answer)
        print()

    if not os.getenv("OPENAI_API_KEY"):

def build_system() -> str:
    """生成系统提示词，里面带上「此刻」的日期与时间

    时间放在这里而不是写死在常量里，是因为 REPL 一轮对话可能跑很久，
    跨天后旧的日期会把模型带偏；需要最新时间时重新调用本函数即可。
    """
    now = datetime.now()
    stamp = f"{now.strftime('%Y-%m-%d %H:%M:%S')} {_WEEKDAYS[now.weekday()]}"
    return (
        f"You are a coding agent at {os.getcwd()}. "
        f"Current date and time: {stamp}. "
        "Use the bash tool to solve tasks. Act, don't explain. "
        "The shell is Windows cmd.exe: use dir / type / echo / findstr / copy / del, "
        "never Unix commands like ls / cat / grep / wc."
    )
def _decode(data: bytes) -> str:
    """按 utf-8 -> gbk 顺序解码命令输出，都失败则替换非法字符"""
    for enc in ("utf-8", "gbk"):
        try:
            return data.decode(enc)
        except UnicodeDecodeError:
            continue
    return data.decode("utf-8", errors="replace")
def _sanitize(text) -> str:
    """去掉 Windows 控制台可能引入的孤立代理字符（如 \\udcxx）

    这类字符会让 openai SDK 在 JSON 序列化时报
    UnicodeEncodeError: surrogates not allowed，必须在发往 API 前清洗。
    """
    text = "" if text is None else str(text)
    return text.encode("utf-8", errors="replace").decode("utf-8")
class BaseTool:
    """工具父类：一个类 = 一个工具

    子类只需要：
        name      -> 暴露给大模型的工具名（必填，因为方法名固定叫 execute）
        execute() -> 真正干活的方法，写清类型注解和 docstring 即可

    schema 全自动：to_openai_schema() 用 generate_tool_description 去解析
    self.execute（bound method，签名里已经没有 self），自动拿到：
        description -> docstring 第一行
        parameters  -> 类型注解 + :param xxx: 说明 + 有无默认值 => required

    想手工覆盖时，才在子类里写同名类属性：
        description -> 覆盖自动生成的描述
        parameters  -> 完全手写 JSON Schema
    """

    name: str = ""
    description: str = ""       # 可选覆盖，留空则用 execute 的 docstring
    parameters: Dict[str, Any] = {}  # 可选覆盖，留空则自动生成

    def execute(self, **kwargs: Any) -> str:
        """执行工具并返回字符串结果（出错就返回 Error: ... 文本）"""
        raise NotImplementedError

    def to_openai_schema(self) -> Dict[str, Any]:
        """转成 OpenAI 规范的 tools 条目

        generate_tool_description 返回的已经是完整的 OpenAI tools 条目
        （type / function.name / function.description / function.parameters），
        这里不重建字典，只在需要覆盖时就地改字段，避免丢掉生成器新增的内容。
        """
        # bound method：self 已绑定，sig/type hints/docstring 都和 s01 的普通函数一样解析
        spec = generate_tool_description(self.execute)
        # 唯一的必改项：bound method 的 __name__ 恒为 "execute"，工具名得用类属性
        spec["function"]["name"] = self.name or type(self).__name__
        if self.description:                 # 有手写描述才覆盖自动生成的 docstring 首行
            spec["function"]["description"] = self.description
        if self.parameters:                  # 有手写 JSON Schema 才覆盖自动生成的参数
            spec["function"]["parameters"] = self.parameters
        return spec
class ToolRegistry:
    """管理所有工具类实例（写法参考 simple_react_agent.py）"""

    def __init__(self):
        self._tools: Dict[str, BaseTool] = {}

    def register(self, tool: BaseTool) -> "ToolRegistry":
        """注册一个工具实例，返回 self 以便链式调用"""
        if not tool.name:
            raise ValueError(f"Tool class {type(tool).__name__} 缺少 name")
        if tool.name in self._tools:
            raise ValueError(f"Duplicate tool name: {tool.name}")
        self._tools[tool.name] = tool
        return self

    def get(self, name: str) -> Optional[BaseTool]:
        """按名字取工具，没有则返回 None"""
        return self._tools.get(name)

    def has(self, name: str) -> bool:
        """工具是否已注册"""
        return name in self._tools

    def execute(self, name: str, params: Dict[str, Any]) -> str:
        """执行工具；未知工具或工具内部异常都转成 Error 文本回给模型"""
        tool = self._tools.get(name)
        if tool is None:
            return f"Error: Unknown tool {name}"
        try:
            return tool.execute(**(params or {}))
        except Exception as e:  # 工具本身炸了（含参数给错）也要把错误喂回模型
            return f"Error: {e}"

    def get_definitions(self) -> List[Dict]:
        """发给大模型的工具描述列表"""
        return [tool.to_openai_schema() for tool in self._tools.values()]

    def as_map(self) -> Dict[str, BaseTool]:
        """工具名 -> 工具实例（保持和 s01 的 TOOL_MAP 同名同用途）"""
        return dict(self._tools)
class RunBashTool(BaseTool):
    """执行一条 shell 命令并返回它的合并输出（Windows cmd.exe）"""

    name = "run_bash"
    # description / parameters 都不写：交给 execute() 的 docstring 与类型注解自动生成

    def __init__(self, workdir: str = None, timeout: int = 120):
        # 共享配置挂在 self 上，这就是「用类写工具」相对散函数的最大好处
        self.workdir = workdir or os.getcwd()
        self.timeout = timeout
        self.dangerous = ["rm -rf /", "sudo", "shutdown", "reboot", "> /dev/", "format "]

    def execute(self, command: str) -> str:
        """执行一条 shell 命令并返回它的输出

        :param command: 要执行的 cmd 命令，例如 dir /b *.py、type a.txt、echo hi > a.txt
        :return: 命令的标准输出与错误输出（合并），无输出时返回占位文本
        """
        if any(d in command for d in self.dangerous):
            return "Error: Dangerous command blocked"
        try:
            r = subprocess.run(
                command,
                shell=True,
                cwd=self.workdir,
                capture_output=True,
                timeout=self.timeout,
            )
            # 拿到原始字节后自己解码：Windows 中文输出多为 GBK，直接按 utf-8 解会得到替换符
            raw = (r.stdout or b"") + (r.stderr or b"")
            out = _decode(raw).strip()
            if not out:
                return f"(no output, exit code {r.returncode})"
            return out[:50000]
        except subprocess.TimeoutExpired:
            return f"Error: Timeout ({self.timeout}s)"
        except (FileNotFoundError, OSError) as e:
            return f"Error: {e}"
def agent_loop(
    messages: list,
    *,
    client=None,                      # 可外部注入：自己的 key / 别的 OpenAI 兼容服务 / 测试用假 client
    registry: ToolRegistry = None,    # 可外部注入：想用自己的工具集就传进来
    tools: list = None,               # 默认由 registry 生成 OpenAI 规范的工具描述
    model: str = None,                # 默认用模块级 MODEL
    max_rounds: int = None,           # 默认用模块级 MAX_ROUNDS
    max_tokens: int = 8000,
    verbose: bool = True,             # False 时全程不 print，方便被别的程序调用
) -> str:
    """反复调用 LLM，直到它不再请求工具为止

    外部依赖全部可注入，所以本文件被 import 后可以直接复用这个函数；
    什么都不传时行为与原来完全一致（用模块级的 client / REGISTRY / MODEL），
    即 agent_loop(messages) 这种老调用方式不受影响。

    :param messages: 会话消息列表，会被原地追加 assistant / tool 消息
    :param client: OpenAI 兼容客户端，不传则用本模块顶部的 client
    :param registry: 工具注册表，不传则用本模块的 REGISTRY
    :param tools: OpenAI 规范的工具描述列表，不传则取 registry.get_definitions()
    :param model: 模型名，不传则用本模块的 MODEL
    :param max_rounds: 安全轮数上限，不传则用本模块的 MAX_ROUNDS
    :param max_tokens: 单次回复的 token 上限
    :param verbose: 是否打印模型回复、命令与工具输出
    :return: 模型最后一轮的纯文本回复；超过轮数上限时返回 [Stopped] 提示
    """
    # 没显式传的，退回模块级默认值；注意 client 与模块级同名，只能走 globals() 取
    if client is None:
        client = globals()["client"]
    if registry is None:
        registry = REGISTRY
    if tools is None:
        tools = registry.get_definitions()
    if model is None:
        model = MODEL
    if max_rounds is None:
        max_rounds = MAX_ROUNDS

    for _ in range(max_rounds):
        response = client.chat.completions.create(
            model=model,
            messages=messages,
            tools=tools,
            tool_choice="auto",
            max_tokens=max_tokens,
        )
        message = response.choices[0].message

        # 1) 追加 assistant 消息（有工具调用时必须带上 tool_calls）
        assistant_msg = {"role": "assistant", "content": _sanitize(message.content or "")}
        if message.tool_calls:
            assistant_msg["tool_calls"] = [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments,
                    },
                }
                for tc in message.tool_calls
            ]
        messages.append(assistant_msg)

        # 2) 没有工具调用 => 循环结束
        if not message.tool_calls:
            return message.content or ""

        # 3) 执行每个工具调用
        for tc in message.tool_calls:
            if verbose and message.content:
                print(message.content)

            func_name = tc.function.name
            try:
                func_args = json.loads(tc.function.arguments or "{}")
            except json.JSONDecodeError:
                func_args = {}

            if not registry.has(func_name):
                output = f"Error: Unknown tool {func_name}"
                if verbose:
                    print(f"\033[31m! {output}\033[0m")
            else:
                if verbose:
                    print(f"\033[33m$ {func_args.get('command', '')}\033[0m")
                output = registry.execute(func_name, func_args)
                if verbose:
                    print(output[:2000])

            # 4) 按 OpenAI 规范把结果回传
            messages.append(
                {
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": _sanitize(output),
                }
            )

    return "[Stopped] Reached max rounds limit."

代码解析

主程序入口

主程序入口
功能概述

该程序实现了一个基于 OpenAI 兼容协议（默认对接 DeepSeek）的 Agent Loop（智能体循环）示例。其核心思想是让大语言模型（LLM）在需要时调用外部工具（此处为执行 Bash 命令），并将工具执行结果回传给模型，循环往复直到模型不再请求工具调用为止。与前一版本（s01）相比，本版本的主要差异在于工具的组织方式：工具不再是模块级函数，而是采用面向对象设计，一个工具对应一个类，通过 BaseTool 基类和 ToolRegistry 注册表进行统一管理，从而提升可扩展性。

入口点与执行流程

程序的入口点是文件末尾的 if __name__ == "__main__" 代码块。它首先检查环境变量 OPENAI_API_KEY 是否存在，若缺失则通过 sys.exit 输出错误并终止。校验通过后调用 main() 函数。main() 是交互式主循环的起点：打印模型、base_url、当前工作目录、可用工具列表等启动信息，然后初始化对话历史 history（首条为 system 消息），进入一个 while True 循环，通过 input() 读取用户输入。若输入为 q、exit 或空字符串则退出；否则将用户输入经 _sanitize 处理后追加到 history，调用 agent_loop(history) 获取回答并打印。

关键初始化步骤与配置

模块加载阶段完成了多项关键初始化。首先是路径处理：将当前文件所在目录插入 sys.path，确保能导入同级的 tool_description_generator 模块；若导入失败则直接退出并提示。其次是控制台兼容处理：在 Windows 上切换代码页到 UTF-8 并启用 ANSI 颜色，同时对 stdout 和 stderr 调用 reconfigure 设置 UTF-8 编码与容错，避免 GBK 控制台下打印中文报错。接着通过 load_dotenv(override=True) 加载 .env 配置，随后创建 OpenAI 客户端，api_key 取自 OPENAI_API_KEY，base_url 默认指向 DeepSeek 的接口（可通过 OPENAI_BASE_URL 覆盖），模型 ID 由 MODEL_ID 决定，默认 deepseek-chat。此外还定义了 MAX_ROUNDS = 40 作为安全上限，防止模型无限调用工具导致费用失控。

主要组件的创建与交互

程序包含几个核心组件。BaseTool 是工具基类，子类只需实现 name 和 execute()，其 to_openai_schema() 方法会调用 generate_tool_description，从 execute 的类型注解和 docstring（:param 形式）自动生成 OpenAI 规范的工具描述，避免手写 schema。RunBashTool 继承 BaseTool，是唯一的实际工具，等价于 s01 中的 run_bash，其共享配置（工作目录、超时、危险命令黑名单）作为实例属性保存在 self 上。ToolRegistry 负责管理工具实例，提供 register、get、execute、get_definitions 等方法，执行出错时返回 "Error: ..." 文本并照常回传给模型。模块级代码创建了 REGISTRY 实例并注册 RunBashTool(os.getcwd())，随后通过 as_map() 生成 TOOL_MAP（工具名到实例的映射）和 get_definitions() 生成 TOOLS（OpenAI 规范的工具描述列表），这两份数据供 agent_loop 使用。SYSTEM 提示词通过 build_system() 在模块加载时生成快照，其中包含中文星期几等信息。

主循环逻辑

agent_loop 的逻辑与 s01 同构：在最多 MAX_ROUNDS 轮内，调用 LLM（传入 messages 和 tools），若响应中没有 tool_calls 则跳出循环并返回最终文本；否则遍历每个 tool_call，从 TOOL_MAP（或 REGISTRY）中按名称取出对应工具实例并执行，将执行结果以 role 为 tool 的消息追加进 messages，然后继续下一轮。这种“模型请求工具—执行工具—回传结果—再次请求模型”的闭环正是 Agent Loop 的核心。外层 main() 的交互循环则负责接收用户输入、维护跨轮次的对话历史，并在每轮结束后打印模型回答。

扩展性与设计要点

该设计强调可扩展性：新增工具只需编写一个继承 BaseTool 的类并调用 REGISTRY.register(XxxTool())，agent_loop 主循环以及 TOOL_MAP、TOOLS 两行常量都无需改动。工具描述由签名和 docstring 自动生成，减少了手写 schema 的维护成本。同时，MAX_ROUNDS 上限、危险命令黑名单、超时控制等机制共同构成了安全与成本防护。整体上，程序将“循环控制”“工具注册”“描述生成”“LLM 调用”解耦为独立组件，结构清晰，便于教学与二次开发。

build_system
核心功能

该函数 build_system 用于动态生成一段系统提示词（system prompt），供编码智能体（coding agent）在对话开始时注入到模型上下文中。提示词内容包含三部分关键信息：当前工作目录、精确到秒的日期时间与星期、以及运行环境的操作约束（Windows cmd.exe 而非 Unix shell）。函数返回一个拼接完成的字符串，供上层调用方直接使用。

设计意图

将时间信息放在函数内部实时计算，而不是固化为模块级常量，是这段代码最核心的设计考量。注释明确说明了原因：REPL 式的一轮对话可能持续很久，若日期在会话开始时被写死，跨天后模型仍会基于过期日期推理，从而产生偏差。通过把 build_system 设计为可重复调用的工厂函数，调用方在需要刷新时间时只需再次调用即可获得最新提示词，实现了时间敏感信息与提示词模板的解耦。

时间格式化细节

函数使用 datetime.now() 获取本地当前时间，并通过 strftime('%Y-%m-%d %H:%M:%S') 格式化为「年-月-日 时:分:秒」的标准形式，便于模型解析。星期部分没有使用 strftime 的 %A 或 %a（其结果受系统 locale 影响，可能输出中文或其他语言），而是通过 now.weekday() 返回的 0-6 整数索引去查表 _WEEKDAYS，从而保证星期名称的输出稳定、可控、与 locale 无关。这是一个值得注意的健壮性处理。

环境信息注入

提示词中通过 os.getcwd() 动态获取当前工作目录并嵌入，使模型了解自己操作的文件系统位置，这对编码智能体执行相对路径操作尤为重要。该值同样是运行时求值，若进程切换了工作目录，重新调用本函数即可反映最新路径。

行为约束设计

提示词后半部分对模型行为做了两类约束。其一是行为风格约束："Act, don't explain."，要求模型直接执行动作而非冗长解释，符合编码智能体的工具调用范式。其二是平台约束：明确声明 shell 为 Windows cmd.exe，并列举了 dir、type、echo、findstr、copy、del 等合法命令，同时显式禁止 ls、cat、grep、wc 等 Unix 命令。这种正反例结合的写法能有效降低模型在 Windows 环境下误用 Unix 命令的概率，是提示词工程中常见的防御性设计。

返回值与纯函数特性

函数无参数、返回 str，除读取系统时间与当前目录外不修改任何外部状态，属于准纯函数（依赖环境但无副作用），易于测试与复用。字符串使用多段 f-string 与普通字符串隐式拼接，可读性较好，但需注意各片段末尾的空格需手动维护以保证最终提示词的分词正确。

潜在改进点

一是 _WEEKDAYS 作为外部依赖未在本函数中体现，若其长度不足 7 或索引语义与 weekday() 不一致会引发错误，建议在模块层做断言或使用常量元组。二是时区问题：datetime.now() 返回本地朴素时间，若部署环境时区不确定，可考虑使用带时区的 datetime.now(timezone.utc).astimezone() 以增强可移植性。三是提示词内容硬编码在函数体内，若未来需要多平台（如 Linux/macOS）支持，可考虑将平台相关片段参数化或抽离为配置。

_decode
核心功能

该函数 _decode 的核心功能是将字节串（bytes）转换为字符串（str），并采用多编码回退策略来应对不同来源的命令输出。设计意图非常明确：在跨平台或跨语言环境下执行系统命令时，输出字节流的编码往往不确定（例如 Windows 中文环境常见 GBK，Linux 常见 UTF-8），因此需要一个健壮的解码函数，尽可能正确地还原文本内容，而不是直接抛出异常导致程序中断。

参数与返回值

函数接收一个名为 data 的 bytes 类型参数，返回一个 str 类型结果。类型注解清晰，便于静态检查和调用方理解接口契约。返回值始终为字符串，不会抛出解码异常，这体现了“永不失败”的容错设计理念。

解码策略与回退顺序

函数使用 for 循环遍历编码元组 ("utf-8", "gbk")，按顺序尝试解码。优先尝试 UTF-8 是因为它是现代系统和网络传输的通用标准，覆盖面最广；若失败则回退到 GBK，主要针对中文 Windows 环境下的命令行输出。这种顺序安排兼顾了通用性与本地化场景。

异常处理机制

在循环内部使用 try/except 捕获 UnicodeDecodeError。当某个编码无法正确解码字节序列时，捕获异常并通过 continue 进入下一次循环尝试下一个编码。这里只捕获特定的 UnicodeDecodeError，而非宽泛的 Exception，属于精确异常处理，避免掩盖其他潜在错误（如类型错误），是良好的编程实践。

最终兜底方案

如果 UTF-8 和 GBK 都解码失败，函数执行最后的 return data.decode("utf-8", errors="replace")。这里使用 errors="replace" 参数，将无法解码的字节替换为替换字符（通常是 U+FFFD），保证函数一定能返回字符串而不抛出异常。这是一种“尽力而为”的降级策略，确保调用方始终能拿到可用的文本结果。

设计权衡与局限

该实现简单高效，适合大多数场景，但也存在一定局限：GBK 解码对任意字节序列的容错性较高，某些本应是其他编码（如 Big5、Shift-JIS）的数据可能被 GBK “错误地成功解码”而产生乱码，因为 GBK 的字节范围较宽。此外，errors="replace" 会丢失原始字节信息，若需保留可考虑使用 errors="backslashreplace"。总体而言，该函数在简洁性、健壮性和实用性之间取得了良好平衡，是处理不确定编码命令输出的合理方案。

_sanitize
核心功能

该函数 _sanitize 的核心职责是对输入文本进行 Unicode 清洗，移除或替换掉可能导致后续 JSON 序列化失败的非法字符，特别是 Windows 控制台环境下可能产生的孤立代理字符（lone surrogates，如 \udcxx 范围内的字符）。设计意图是作为一道防御性屏障，在文本发往 OpenAI API 之前保证其可被安全地 UTF-8 编码和 JSON 序列化。

问题背景

孤立代理字符是 UTF-16 编码体系中的残留产物。Unicode 代理区（U+D800 到 U+DFFF）本应成对出现以表示增补平面字符，但 Windows 控制台、剪贴板或某些输入源可能产生未配对的单个代理码点。Python 字符串可以容纳这类字符，但当 openai SDK 尝试将其序列化为 JSON 并编码为 UTF-8 时，会抛出 UnicodeEncodeError: surrogates not allowed，导致请求失败。该函数正是为规避这一运行时错误而存在。

输入归一化

函数第一行 text = "" if text is None else str(text) 完成了输入的类型归一化。当传入 None 时返回空字符串，避免后续对 None 调用 encode 引发 AttributeError；对其他任意类型则强制转换为字符串，使函数具备一定的容错性，可接受非字符串输入而不崩溃。

核心实现

关键逻辑是 text.encode("utf-8", errors="replace").decode("utf-8") 这一编码-解码往返操作。encode 阶段使用 errors="replace" 参数，意味着遇到无法编码为 UTF-8 的字符（即孤立代理）时，不会抛出异常，而是替换为 UTF-8 的替换字符 U+FFFD（显示为 �）。随后 decode 将字节流还原为合法的 Python 字符串。经过这一往返，所有非法代理字符都被替换为安全的替换符，字符串变为可安全序列化的形式。

技术要点

使用 errors="replace" 而非 errors="ignore" 是有意为之：替换策略保留了"此处存在异常字符"的信息，便于调试和问题追溯，而忽略策略会静默丢弃字符，可能掩盖数据质量问题。此外，该实现利用了 UTF-8 编码器对代理字符的严格校验特性——合法的增补平面字符（如 emoji）会被正确编码为四字节序列并在解码后完整保留，只有真正非法的孤立代理才会被替换，因此不会误伤正常的多字节字符。

返回值与副作用

函数返回清洗后的字符串，保证其可被 UTF-8 编码，从而可安全通过 JSON 序列化。函数为纯函数式设计，不修改外部状态，无副作用，可安全地在多处调用。对于已经是合法 UTF-8 的输入，编码-解码往返是幂等的，不会改变内容。

性能考量

每次调用都会执行一次完整的编码与解码，对于超长文本会产生额外的内存分配与 CPU 开销。在需要高频清洗大量文本的场景下，可考虑先做快速检测（如尝试编码并捕获异常）以避免不必要的往返，但就当前用途（API 请求前的单次清洗）而言，该开销可以忽略，实现简洁性优先是合理取舍。

BaseTool
核心功能

BaseTool 是一个工具基类，采用“一个类对应一个工具”的设计模式，用于将 Python 类封装为符合 OpenAI Function Calling 规范的工具定义。其核心目标是把工具的描述、参数结构等信息自动化地从 execute 方法的签名、类型注解和 docstring 中提取出来，减少手写 JSON Schema 的重复劳动，同时保留手工覆盖的能力。

设计意图

该基类通过约定优于配置的思路，规定子类只需提供 name 类属性和实现 execute 方法即可。工具名固定由类属性 name 提供，因为方法名恒为 execute，无法从方法名推导工具名。描述和参数默认从 execute 的 docstring 与类型注解自动生成，只有在需要特殊定制时才通过同名类属性覆盖，从而在自动化与可控性之间取得平衡。

类属性定义

name 为工具名，默认空字符串，在生成 schema 时若为空则回退到类名。description 为可选的描述覆盖项，留空则使用 execute 的 docstring 首行。parameters 为可选的 JSON Schema 覆盖项，留空则自动生成。三者均作为类属性存在，子类可按需覆盖。

execute 方法

这是子类必须实现的抽象方法，签名使用 **kwargs 接收任意关键字参数，返回字符串结果，约定出错时返回以 Error: 开头的文本。基类中抛出 NotImplementedError，强制子类实现，起到接口约束作用。

to_openai_schema 方法

这是核心转换方法，负责生成 OpenAI tools 条目。它调用 generate_tool_description 解析 self.execute 这个绑定方法（bound method），由于 self 已绑定，签名中不再包含 self，因此可以像解析普通函数一样获取类型注解、docstring 和参数信息。生成结果本身已是完整的 OpenAI tools 条目结构，包含 type、function.name、function.description、function.parameters 等字段。

名称覆盖逻辑

由于绑定方法的 __name__ 恒为 execute，自动生成的工具名会是 execute，因此必须用类属性 name 覆盖。代码使用 self.name or type(self).__name__，即优先使用 name，为空时回退到类名，保证工具名始终有效且唯一可读。

描述与参数覆盖逻辑

采用“就地修改”而非重建字典的策略：仅当 self.description 非空时才覆盖 function.description，仅当 self.parameters 非空时才覆盖 function.parameters。这种设计避免了重建字典时丢失生成器可能新增的字段，具有良好的向前兼容性。

技术要点

一是利用绑定方法的签名解析特性，使 self 自动被排除，简化了参数提取；二是通过类属性实现可选的覆盖机制，兼顾自动化与灵活性；三是就地修改字典而非重建，保证与生成器输出结构的兼容性；四是用 name 回退到类名的方式处理工具命名，避免名称缺失。整体体现了约定式接口设计与元数据驱动 schema 生成的思想。

ToolRegistry
核心功能

ToolRegistry 是一个工具注册中心类，用于集中管理所有工具类实例。它承担了工具注册、查找、执行和描述导出等职责，是构建 ReAct 风格 Agent 时常用的基础设施组件。设计意图是将工具的注册与调用解耦，让 Agent 主循环只需通过工具名和参数字典即可调用任意工具，同时把工具执行过程中的异常统一转换为文本错误信息回传给大模型，避免异常中断整个推理流程。

内部状态

类内部使用私有字典 _tools 存储工具，键为工具名称字符串，值为 BaseTool 类型实例。使用字典结构保证了按名称查找的时间复杂度为 O(1)，同时天然支持判断工具是否存在以及遍历所有工具。类型注解 Dict[str, BaseTool] 明确了存储结构，便于静态检查和阅读。

构造函数

__init__ 方法仅初始化空字典 _tools，不接收任何参数。这种设计让注册表本身保持轻量，工具通过后续的 register 方法逐个注入，符合依赖注入和可扩展的设计思路。

注册方法 register

register 方法接收一个 BaseTool 实例，先校验工具是否具有非空的 name 属性，若缺失则抛出 ValueError 并附带具体类名，便于定位问题。随后检查名称是否重复，重复注册同样抛出 ValueError，防止同名工具覆盖导致调用歧义。校验通过后将工具存入字典，并返回 self，从而支持链式调用写法，例如 registry.register(a).register(b)，使注册代码更简洁。

查询方法 get 与 has

get 方法通过字典的 get 按名称取工具，未找到时返回 None 而非抛异常，调用方可以自行决定如何处理缺失情况。has 方法通过 in 运算符判断工具是否已注册，返回布尔值。两者配合提供了安全的查询能力，避免直接访问内部字典。

执行方法 execute

execute 是核心方法，接收工具名和参数字典。首先查找工具，若不存在则返回格式化的错误文本 "Error: Unknown tool {name}"，而不是抛出异常。找到工具后，使用 params or {} 处理参数为 None 的情况，再通过关键字参数展开调用 tool.execute。整个调用包裹在 try/except 中，捕获所有异常并转换为 "Error: {e}" 文本返回。这种设计的关键意图是：无论工具不存在、参数错误还是工具内部逻辑崩溃，都以文本形式反馈给大模型，让模型有机会根据错误信息调整后续动作，而不是让整个 Agent 流程因异常而终止，这是 ReAct 类 Agent 容错设计的重要一环。

描述导出 get_definitions

get_definitions 遍历所有已注册工具，调用每个工具的 to_openai_schema 方法，返回符合 OpenAI 函数调用格式的工具描述列表。该方法将工具注册表与具体的大模型 API 格式对接，使 Agent 可以把工具清单直接放入请求中，实现工具能力的动态暴露。

映射导出 as_map

as_map 返回内部字典的浅拷贝，避免外部直接修改内部状态。注释说明其用途与 simple_react_agent.py 中的 TOOL_MAP 保持一致，便于与既有代码风格兼容，也方便需要以字典形式访问工具集合的场景使用。

设计要点总结

整体设计体现了几个技术要点：一是通过名称到实例的映射实现工具的动态注册与查找；二是链式调用提升注册代码的可读性；三是异常统一转为文本错误，保证 Agent 循环的健壮性；四是通过 to_openai_schema 与 as_map 分别对接大模型协议和传统字典用法，兼顾扩展性与兼容性。该类职责单一、边界清晰，是 Agent 工具层良好的封装范例。

RunBashTool
核心功能

RunBashTool 是一个继承自 BaseTool 的工具类，用于在 Windows cmd.exe 环境下执行单条 shell 命令，并将标准输出与标准错误合并后返回给调用方。它的设计意图是为上层（如 Agent 或工具调用框架）提供一个受控、可配置、带安全防护的命令执行入口，同时通过类封装共享工作目录、超时时间等配置。

类属性与元信息

类中只显式声明了 name = "run_bash"，而 description 和 parameters 故意留空。代码注释说明这些元信息由 execute() 的 docstring 与类型注解自动生成，这是一种“约定优于配置”的设计，减少重复描述，保证文档与签名一致，也便于框架通过反射提取工具描述和参数 schema。

构造函数

__init__ 接收 workdir 和 timeout 两个可选参数。workdir 为空时回退到 os.getcwd()，timeout 默认 120 秒。构造函数还初始化了 dangerous 列表，包含 rm -rf /、sudo、shutdown、reboot、> /dev/、format 等危险命令片段。注释强调“共享配置挂在 self 上”是用类写工具相对散函数的最大好处，即配置与状态可复用、可注入。

安全防护

execute 首先用 any(d in command for d in self.dangerous) 做子串匹配，命中即返回 "Error: Dangerous command blocked"。这是一种轻量级黑名单防护，实现简单但存在明显局限：子串匹配容易被绕过（如空格变形、变量拼接、大小写、编码混淆），也无法覆盖所有危险操作，因此只能作为第一道粗筛，不能替代真正的沙箱或权限隔离。

命令执行

核心调用是 subprocess.run，参数包括 shell=True、cwd=self.workdir、capture_output=True、timeout=self.timeout。shell=True 使得可以直接执行 cmd 内建命令和管道重定向，符合工具定位；cwd 保证命令在指定工作目录下运行；capture_output 同时捕获 stdout 和 stderr；timeout 防止命令长时间挂起。

输出解码

代码将 r.stdout 与 r.stderr 以字节形式拼接（对 None 用 b"" 兜底），再交给 _decode(raw) 解码。注释指出 Windows 中文输出多为 GBK，直接按 utf-8 解码会产生替换符，因此自行解码是必要的兼容处理。这体现了对跨编码环境的实际考量，但 _decode 的具体实现未在片段中给出，其健壮性取决于该函数是否做了多编码回退。

结果处理

解码后调用 strip() 去除首尾空白。若结果为空，返回带退出码的占位文本 "(no output, exit code N)"，便于调用方区分“无输出”和“执行失败”。非空结果则截断到前 50000 个字符，防止超长输出撑爆上下文或内存，这是一种务实的资源保护措施。

异常处理

显式捕获 subprocess.TimeoutExpired，返回带超时秒数的错误文本；捕获 FileNotFoundError 和 OSError，返回 "Error: {e}"。异常被转换为字符串返回值而非抛出，符合工具调用场景下“以文本反馈结果”的契约，使上层无需额外处理异常分支。不过未捕获其他异常（如解码异常、权限相关异常的子类差异），覆盖面可进一步完善。

设计评价

整体结构清晰，职责单一，配置可注入，安全与资源限制都有考虑，适合作为 Agent 工具集中的一个基础执行器。主要改进空间在于：危险命令检测应改为更结构化的解析或白名单/沙箱机制；_decode 需明确多编码回退策略；截断 50000 字符时可附带截断提示；以及可考虑对 shell=True 的风险做更明确的文档说明。

agent_loop
核心功能

agent_loop 实现了一个典型的 ReAct 风格智能体主循环：反复调用大语言模型（LLM），解析模型返回的工具调用请求，执行对应工具，把结果回填到消息列表，再继续下一轮，直到模型不再请求工具或达到轮数上限。函数最终返回模型最后一轮的纯文本回复，若超过 max_rounds 则返回固定的停止提示字符串。

设计意图

该函数的核心设计目标是可复用性与可测试性。通过将所有外部依赖（client、registry、tools、model、max_rounds）设计为关键字参数并允许为 None，函数在无参调用时回退到模块级默认值，从而保持与旧调用方式 agent_loop(messages) 的完全兼容；同时允许调用方注入自定义客户端、工具集或测试替身，便于单元测试和跨服务复用。verbose 参数则让函数既可作为命令行工具直接运行，也可作为库被其他程序静默调用。

参数默认值回退机制

函数开头对每个可选参数做 None 判断并回退到模块级变量。值得注意的是 client 参数与模块级变量同名，函数内部直接引用 client 会命中局部参数而非全局变量，因此代码显式使用 globals()["client"] 来获取模块级客户端，这是一个容易出错但处理正确的细节。registry 回退到 REGISTRY，tools 在未提供时通过 registry.get_definitions() 动态生成 OpenAI 规范的工具描述，model 与 max_rounds 分别回退到 MODEL 和 MAX_ROUNDS。

主循环结构

使用 for _ in range(max_rounds) 作为安全上限，避免模型陷入无限工具调用循环。每轮调用 client.chat.completions.create，传入 model、messages、tools、tool_choice="auto" 和 max_tokens，其中 tool_choice="auto" 表示由模型自行决定是否调用工具。循环体按“追加 assistant 消息 → 判断是否结束 → 执行工具 → 回填结果”的顺序组织，逻辑清晰。

assistant 消息构造

每轮先将模型回复追加到 messages。content 通过 _sanitize 处理（推测用于清理非法字符或编码问题），并在为空时用空字符串兜底。若存在 tool_calls，则按 OpenAI 规范将其转换为包含 id、type、function（name 与 arguments）的字典列表一并写入 assistant 消息。这一步是必须的，因为后续 tool 消息需要通过 tool_call_id 与之对应，缺失会导致 API 报错。

循环终止条件

在追加 assistant 消息之后立即检查 message.tool_calls：若为空，说明模型已给出最终答复，函数直接返回 message.content 或空字符串。这一判断放在工具执行之前，保证纯文本回复不会被误当作需要继续处理的中间状态。

工具执行与容错

对每个 tool_call，先尝试用 json.loads 解析 arguments，解析失败时回退为空字典 {}，避免因模型输出非法 JSON 导致整个循环崩溃。随后通过 registry.has 检查工具是否存在：不存在时构造 "Error: Unknown tool ..." 的错误文本作为工具输出回传，让模型有机会自我纠正；存在时调用 registry.execute 执行并获取输出。verbose 模式下会打印模型文本、命令（取 func_args 中的 command 字段）以及截断到 2000 字符的工具输出，并使用 ANSI 颜色码区分提示信息。

结果回填规范

每个工具执行结果都以 {"role": "tool", "tool_call_id": tc.id, "content": _sanitize(output)} 的形式追加到 messages，严格遵循 OpenAI 的工具调用协议。tool_call_id 与 assistant 消息中的 id 一一对应，确保多工具并行调用时结果能正确归属。输出同样经过 _sanitize 处理，保证消息内容符合 API 要求。

轮数上限处理

若 for 循环正常结束仍未返回（即达到 max_rounds 仍未停止），函数返回 "[Stopped] Reached max rounds limit."。这是一种防御性设计，防止模型持续请求工具导致资源耗尽或死循环。

副作用与可变性

messages 参数会被原地修改（追加 assistant 与 tool 消息），调用方传入的列表在函数返回后包含完整对话历史，这既是特性也需调用方注意。函数本身除打印（verbose 时）外无其他全局副作用，整体是同步阻塞式实现，未涉及并发或异步处理。
